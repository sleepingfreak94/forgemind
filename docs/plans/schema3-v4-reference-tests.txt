const legacyDDL = `
  CREATE TABLE identity (singleton INTEGER PRIMARY KEY CHECK(singleton = 1), project_id TEXT NOT NULL, workspace TEXT NOT NULL) STRICT;
  CREATE TABLE records (id TEXT PRIMARY KEY, digest TEXT NOT NULL, kind TEXT NOT NULL,
    state TEXT NOT NULL, expires_at INTEGER, body TEXT) STRICT;
  CREATE VIRTUAL TABLE memory_fts USING fts5(record_id UNINDEXED, title, text, tokenize='unicode61');
  CREATE TABLE audit (sequence INTEGER PRIMARY KEY, action TEXT NOT NULL, record_id TEXT NOT NULL, at INTEGER NOT NULL) STRICT;
  CREATE TABLE deleted_sources (path TEXT NOT NULL, sha256 TEXT NOT NULL, PRIMARY KEY(path, sha256)) STRICT;
  INSERT INTO memory_fts(memory_fts, rank) VALUES ('secure-delete', 1);
`;
const historyDDL = `CREATE TABLE run_history (sequence INTEGER PRIMARY KEY AUTOINCREMENT, id TEXT NOT NULL UNIQUE,
  digest TEXT NOT NULL, body TEXT, recorded_at INTEGER NOT NULL,
  revoked INTEGER NOT NULL DEFAULT 0 CHECK(revoked IN (0,1))) STRICT;`;
function store(f: ReturnType<typeof fixture>) {
  const directory = join(f.base, 'schema-state'), path = databasePath(directory, f.root, f.projectId);
  mkdirSync(directory, { mode: 0o700 }); mkdirSync(dirname(path), { mode: 0o700 });
  closeSync(openSync(path, 'wx', 0o600));
  return { directory, path };
}
function legacy(f: ReturnType<typeof fixture>, version = 1, projectId = f.projectId) {
  const selected = store(f);
  inspect(selected.path, db => {
    db.exec(legacyDDL);
    db.prepare('INSERT INTO identity VALUES (1, ?, ?)').run(projectId, f.root);
    db.prepare('INSERT INTO records VALUES (?, ?, ?, ?, ?, ?)').run('legacy', 'digest', 'decision', 'active', null, '{"legacy":true}');
    db.prepare('INSERT INTO memory_fts(record_id, title, text) VALUES (?, ?, ?)').run('legacy', 'SQLite storage', 'Durable decisions');
    db.exec("INSERT INTO audit VALUES (7, 'create', 'legacy', 1000); INSERT INTO deleted_sources VALUES ('docs/old.md', 'old-sha');");
    db.exec(`PRAGMA user_version = ${version}`);
  });
  return selected;
}
function inspect<T>(path: string, work: (db: DatabaseSync) => T): T {
  const db = new DatabaseSync(path);
  try { return work(db); } finally { db.close(); }
}
function contents(db: DatabaseSync) {
  return ['identity', 'records', 'memory_fts', 'audit', 'deleted_sources'].map(table => db.prepare(`SELECT * FROM ${table} ORDER BY rowid`).all());
}
function snapshot(path: string) {
  return inspect(path, db => ({
    version: db.prepare('PRAGMA user_version').get()?.user_version,
    schema: db.prepare('SELECT type, name, tbl_name, sql FROM sqlite_schema ORDER BY type, name').all(),
    contents: contents(db),
    history: db.prepare("SELECT 1 FROM sqlite_schema WHERE name = 'run_history'").get() ? db.prepare('SELECT * FROM run_history ORDER BY sequence').all() : null
  }));
}
function sidecars(path: string) {
  return ['-journal', '-wal', '-shm'].map(suffix => existsSync(path + suffix));
}
function rejectsUnchanged(f: ReturnType<typeof fixture>, selected: ReturnType<typeof store>, pattern: RegExp) {
  const before = snapshot(selected.path), bytes = readFileSync(selected.path), files = sidecars(selected.path);
  assert.throws(() => new MemoryDatabase(selected.directory, f.root, f.projectId), pattern);
  assert.deepEqual(readFileSync(selected.path), bytes);
  assert.deepEqual(snapshot(selected.path), before);
  assert.deepEqual(sidecars(selected.path), files);
  inspect(selected.path, db => db.exec('BEGIN EXCLUSIVE; ROLLBACK;'));
}
function rawFiles(path: string) {
  return ['', '-journal', '-wal', '-shm'].map(suffix => existsSync(path + suffix) ? readFileSync(path + suffix) : null);
}
function rejectsRaw(f: ReturnType<typeof fixture>, selected: ReturnType<typeof store>, pattern: RegExp, probes = 0) {
  const before = rawFiles(selected.path), connections = new Set<DatabaseSync>(), prepare = DatabaseSync.prototype.prepare;
  const spy = mock.method(DatabaseSync.prototype, 'prepare', function(this: DatabaseSync, sql: string) {
    connections.add(this); return prepare.call(this, sql);
  });
  try {
    assert.throws(() => new MemoryDatabase(selected.directory, f.root, f.projectId), pattern);
    assert.deepEqual(rawFiles(selected.path), before);
    assert.equal(connections.size, probes);
    for (const db of connections) assert.equal(db.isOpen, false);
  } finally { spy.mock.restore(); }
}

    const selected = legacy(f, version);
    inspect(selected.path, db => {
      for (const state of ['expired', 'revoked', 'superseded']) {
        db.prepare('INSERT INTO records VALUES (?, ?, ?, ?, ?, ?)').run(state, `digest-${state}`, 'decision', state, 123, null);
      }
      if (version === 2) {
        db.exec(historyDDL);
        db.exec(`INSERT INTO run_history(sequence, id, digest, body, recorded_at, revoked) VALUES
          (2, 'retained', 'd1', 'original body', 1001, 0),
          (5, 'redacted', 'd2', NULL, 1002, 0), (9, 'revoked', 'd3', NULL, 1003, 1),
          (20, 'removed', 'd4', NULL, 1004, 0);
          DELETE FROM run_history WHERE sequence = 20;`);
      }
    });
    const before = snapshot(selected.path);
    const oldSequence = version === 2 ? inspect(selected.path, db => db.prepare('SELECT * FROM sqlite_sequence').all()) : null;
    const oldHistoryRoot = version === 2 ? inspect(selected.path, db => db.prepare("SELECT rootpage, sql FROM sqlite_schema WHERE name = 'run_history'").get()) : null;
    const memory = new MemoryDatabase(selected.directory, f.root, f.projectId);
    try {
      assert.equal(memory.db.prepare('PRAGMA user_version').get()?.user_version, 3);
      assert.deepEqual(contents(memory.db), before.contents);
      assert.equal(memory.db.prepare("SELECT record_id FROM memory_fts WHERE memory_fts MATCH 'SQLite'").get()?.record_id, 'legacy');
      if (version === 2) {
        assert.deepEqual(memory.db.prepare('SELECT * FROM run_history ORDER BY sequence').all(), before.history);
        assert.deepEqual(memory.db.prepare('SELECT * FROM sqlite_sequence').all(), oldSequence);
        assert.deepEqual(memory.db.prepare("SELECT rootpage, sql FROM sqlite_schema WHERE name = 'run_history'").get(), oldHistoryRoot);
        memory.db.exec("INSERT INTO run_history(id, digest, recorded_at) VALUES ('next', 'next-digest', 1005)");
        assert.equal(memory.db.prepare("SELECT sequence FROM run_history WHERE id = 'next'").get()?.sequence, 21);
      } else assert.deepEqual(memory.db.prepare('SELECT * FROM run_history').all(), []);
    } finally { memory.close(); }
    const migrated = snapshot(selected.path), reopened = new MemoryDatabase(selected.directory, f.root, f.projectId);
    try { assert.deepEqual(snapshot(selected.path), migrated); } finally { reopened.close(); }
  } finally { f.dispose(); }
});

  const f = fixture();

  try {
    owner = new MemoryDatabase(f.directory, f.root, f.projectId);
    const db = owner.db;
    for (const name of ['weekly_control', 'weekly_summaries', 'weekly_dependencies']) {
      assert.equal(db.prepare('SELECT strict FROM pragma_table_list WHERE name = ?').get(name)?.strict, 1);
    }
    assert.deepEqual({ ...db.prepare('SELECT * FROM weekly_control').get() },
      { singleton: 1, watermark: 0, last_due: 0, fence: 0, lease_expiry: 0, lease_owner: null });
    for (const field of ['watermark', 'last_due', 'fence', 'lease_expiry']) {
      const info = db.prepare('SELECT * FROM pragma_table_info(?) WHERE name = ?').get('weekly_control', field);
      assert.equal(info?.notnull, 1); assert.equal(info?.dflt_value, '0');
      assert.throws(() => db.exec(`UPDATE weekly_control SET ${field} = -1`));
      assert.throws(() => db.exec(`UPDATE weekly_control SET ${field} = NULL`));
      assert.throws(() => db.exec(`UPDATE weekly_control SET ${field} = 'bad'`));

    assert.throws(() => db.exec('INSERT INTO weekly_control(singleton) VALUES (2)'));
    const insert = db.prepare('INSERT INTO weekly_summaries VALUES (?, ?, ?, ?, ?, ?, ?, ?)');
    for (const values of [
      ['negative', 'd', -1, 1, 0, 0, 'complete', null], ['upper', 'd', 2, 1, 1, 0, 'complete', null],
      ['cursor-low', 'd', 1, 2, 0, 0, 'complete', null], ['cursor-high', 'd', 1, 2, 3, 0, 'complete', null],
      ['due', 'd', 0, 0, 0, -1, 'complete', null], ['state', 'd', 0, 0, 0, 0, 'bad', null]
    ] as const) assert.throws(() => insert.run(...values));
    insert.run('pending', 'digest', 0, 5, 0, 0, 'pending', 'body');
    assert.throws(() => insert.run('second', 'digest', 0, 5, 0, 0, 'pending', null));
    insert.run('complete', 'digest', 0, 5, 5, 0, 'complete', null);
    assert.throws(() => db.exec("INSERT INTO weekly_dependencies VALUES ('missing', 'run', 'digest')"));

    assert.throws(() => db.exec("INSERT INTO weekly_dependencies VALUES ('pending', 'other', NULL)"));
    assert.deepEqual(db.prepare('PRAGMA index_info(weekly_dependencies_run)').all().map(row => row.name), ['run_id', 'run_digest']);
    const pending = db.prepare("SELECT * FROM pragma_index_list('weekly_summaries') WHERE name = 'weekly_one_pending'").get();
    assert.equal(pending?.unique, 1); assert.equal(pending?.partial, 1);
    const foreign = db.prepare('PRAGMA foreign_key_list(weekly_dependencies)').get();
    assert.equal(foreign?.table, 'weekly_summaries'); assert.equal(foreign?.from, 'summary_id'); assert.equal(foreign?.to, 'id');
    assert.equal(db.prepare("SELECT tbl_name FROM sqlite_schema WHERE type = 'trigger' AND name = 'weekly_history_invalidation'").get()?.tbl_name, 'run_history');
  } finally { try { owner?.close(); } finally { f.dispose(); } }

test('private paths, links and sidecars remain protected', () => {
  const f = fixture(); try {
    const selected = legacy(f);
    assert.equal(statSync(selected.path).mode & 0o777, 0o600);
    assert.equal(statSync(selected.directory).mode & 0o777, 0o700);
    const bytes = readFileSync(selected.path);
    if (typeof process.getuid === 'function') {
      for (const path of [selected.directory, dirname(selected.path), selected.path]) {
        const mode = statSync(path).mode & 0o777; chmodSync(path, mode | 0o040);
        try { assert.throws(() => new MemoryDatabase(selected.directory, f.root, f.projectId), /private/); }
        finally { chmodSync(path, mode); }
      }
    }
    const alias = join(f.base, 'storage-alias'); symlinkSync(selected.directory, alias);
    assert.throws(() => new MemoryDatabase(alias, f.root, f.projectId), /canonical/);
    const linked = join(f.base, 'database-link'); linkSync(selected.path, linked);
    try { assert.throws(() => new MemoryDatabase(selected.directory, f.root, f.projectId), /unsafe/); }
    finally { unlinkSync(linked); }
    for (const suffix of ['-journal', '-wal', '-shm']) {
      symlinkSync(selected.path, selected.path + suffix);
      try { assert.throws(() => new MemoryDatabase(selected.directory, f.root, f.projectId), /unsafe/); }
      finally { unlinkSync(selected.path + suffix); }
    }
    assert.deepEqual(readFileSync(selected.path), bytes);
  } finally { f.dispose(); }
});

for (const change of ['revoked = 1', 'body = NULL', "body = 'changed'", "digest = 'changed'", 'revoked = revoked', 'body = body']) {
  test(`history update ${change} atomically invalidates all dependent summaries`, () => {
    const f = fixture(); let owner: MemoryDatabase | undefined;
    try {
      owner = new MemoryDatabase(f.directory, f.root, f.projectId);
      const db = owner.db, revoked = change === 'revoked = revoked' ? 1 : 0, body = change === 'body = body' ? null : 'original';
      db.prepare('INSERT INTO run_history(id, digest, body, recorded_at, revoked) VALUES (?, ?, ?, ?, ?)').run('run', 'd', body, 1000, revoked);
      db.exec(`INSERT INTO weekly_summaries VALUES
        ('a', 'd', 0, 1, 0, 0, 'pending', 'secret-a'),
        ('b', 'd', 0, 1, 1, 0, 'complete', 'secret-b'),
        ('unrelated', 'd', 0, 1, 1, 0, 'complete', 'keep');
        INSERT INTO weekly_dependencies VALUES ('a', 'run', 'd'), ('b', 'run', 'd'), ('unrelated', 'other', 'd');`);
      const summaries = () => db.prepare('SELECT * FROM weekly_summaries ORDER BY id').all();
      const before = summaries(), history = db.prepare('SELECT * FROM run_history').all();
      const links = db.prepare('SELECT * FROM weekly_dependencies ORDER BY summary_id').all();
      const verify = () => {
        const rows = summaries();
        for (const row of rows.filter(row => row.id !== 'unrelated')) { assert.equal(row.body, null); assert.equal(row.state, 'invalidated'); }
        assert.deepEqual(rows.find(row => row.id === 'unrelated'), before.find(row => row.id === 'unrelated'));
        assert.deepEqual(db.prepare('SELECT * FROM weekly_dependencies ORDER BY summary_id').all(), links);
        assert.equal(db.prepare('SELECT count(*) AS n FROM run_history').get()?.n, 1);
      };
      const rollback = new Error('rollback');
      assert.throws(() => owner!.transaction(() => { db.exec(`UPDATE run_history SET ${change} WHERE id = 'run'`); verify(); throw rollback; }), error => error === rollback);
      assert.deepEqual(summaries(), before); assert.deepEqual(db.prepare('SELECT * FROM run_history').all(), history);
      owner.transaction(() => db.exec(`UPDATE run_history SET ${change} WHERE id = 'run'`)); verify();
    } finally { try { owner?.close(); } finally { f.dispose(); } }
  });
}
